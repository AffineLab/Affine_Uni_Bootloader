cmake_minimum_required(VERSION 3.20)
include(${CMAKE_CURRENT_LIST_DIR}/../../cmake/pico-sdk.cmake)
if(PICO_NO_FLASH OR PICO_COPY_TO_RAM OR PICO_NO_PICOTOOL OR PICO_NO_UF2)
    message(FATAL_ERROR "RP2350 target requires flash execution and SDK picotool/UF2 outputs")
endif()
foreach(_pair "PICO_PLATFORM;rp2350-arm-s" "PICO_BOARD;pico2")
    list(GET _pair 0 _key)
    list(GET _pair 1 _value)
    if(DEFINED ${_key} AND NOT "${${_key}}" STREQUAL "${_value}")
        message(FATAL_ERROR "${_key} must be ${_value}")
    endif()
    set(${_key} "${_value}" CACHE STRING "RP2350 reference configuration" FORCE)
endforeach()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(affine_rp2350 C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
# Require an installed tool; the SDK must never silently fetch one here.
find_package(picotool 2.3.1 EXACT CONFIG REQUIRED NO_CMAKE_FIND_ROOT_PATH)
pico_sdk_init()
get_filename_component(BOOT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(_pair "ALLOW_UNSIGNED_UPDATES;1" "REQUIRE_SIGNED_UPDATES;0"
        "ALLOW_OPTIONAL_ANTI_ROLLBACK;1" "REQUIRE_ANTI_ROLLBACK_FOR_SIGNED;0"
        "PRESERVE_SECURITY_STATE_ON_CLEAR_METADATA;1")
    list(GET _pair 0 _policy)
    list(GET _pair 1 _default)
    set(AFFINE_BOOT_${_policy} ${_default} CACHE BOOL "Shared boot policy: ${_policy}")
    list(APPEND _policy_defines "AFFINE_BOOT_${_policy}=$<BOOL:${AFFINE_BOOT_${_policy}}>")
endforeach()
add_executable(bootloader_rp2350 main.c usb.c usb_descriptors.c
    boot_platform_rp2350.c flash.c
    ${BOOT_ROOT}/targets/common/boot_target_common.c
    ${BOOT_ROOT}/src/bootloader.c ${BOOT_ROOT}/src/boot_protocol.c
    ${BOOT_ROOT}/src/boot_image.c ${BOOT_ROOT}/src/boot_security.c
    ${BOOT_ROOT}/src/boot_crypto.c)
target_include_directories(bootloader_rp2350 PRIVATE . ${BOOT_ROOT} ${BOOT_ROOT}/include)
target_compile_options(bootloader_rp2350 PRIVATE -Os -Wall -Wextra)
target_compile_definitions(bootloader_rp2350 PRIVATE ${_policy_defines})
set(AFFINE_BOOT_KEYS_HEADER "" CACHE FILEPATH "Optional generated key override header")
if(AFFINE_BOOT_KEYS_HEADER)
    get_filename_component(_keys "${AFFINE_BOOT_KEYS_HEADER}" ABSOLUTE)
    if(NOT EXISTS "${_keys}")
        message(FATAL_ERROR "Missing key override: ${_keys}")
    endif()
    target_compile_definitions(bootloader_rp2350 PRIVATE "AFFINE_BOOT_KEYS_OVERRIDE_HEADER=\"${_keys}\"")
endif()
target_compile_definitions(bootloader_rp2350 PRIVATE PICO_STACK_SIZE=0x2000)
# Core 1 is unused. Combine both 4 KiB scratch banks for the core 0 stack.
pico_set_linker_script_var(bootloader_rp2350 SCRATCH_X_LENGTH 0)
pico_set_linker_script_var(bootloader_rp2350 SCRATCH_Y_ORIGIN 0x20080000)
pico_set_linker_script_var(bootloader_rp2350 SCRATCH_Y_LENGTH 0x2000)
target_link_libraries(bootloader_rp2350 pico_stdlib tinyusb_device hardware_flash hardware_watchdog)
pico_enable_stdio_usb(bootloader_rp2350 0)
pico_enable_stdio_uart(bootloader_rp2350 0)
pico_add_linker_script_override_path(bootloader_rp2350 ${CMAKE_CURRENT_LIST_DIR}/boot_link FILES pico_flash_region.ld)
# Flash capacity is the common 2 MiB contract, independent of Pico 2's 4 MiB.
target_compile_definitions(bootloader_rp2350 PRIVATE PICO_FLASH_SIZE_BYTES=2097152)
pico_set_uf2_family(bootloader_rp2350 rp2350-arm-s)
pico_add_extra_outputs(bootloader_rp2350)

option(AFFINE_RP2350_BUILD_TEST_APP "Build the offset jump/re-entry test app" ON)
if(AFFINE_RP2350_BUILD_TEST_APP)
    add_subdirectory(test_app)
endif()
